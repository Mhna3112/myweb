async function probeHead(url){
  try{
    const ctl = new AbortController(); const t = setTimeout(()=>ctl.abort(), 3500);
    const r = await fetch(url, {method:'HEAD', mode:'cors', signal:ctl.signal, redirect:'follow'});
    clearTimeout(t);
    const mime = (r.headers.get('content-type')||'').split(';')[0].trim().toLowerCase();
    if(/^image\//.test(mime)) return {type:'image', mime};
    if(/^video\//.test(mime)) return {type:'video', mime};
    if(/^audio\//.test(mime)) return {type:'audio', mime};
    return null;
  }catch(e){ return null; }
}

/* L3 — Magic bytes: chữ ký nhị phân thật (cần CORS; dùng Range cho nhẹ) */
async function probeMagic(url){
  try{
    const ctl = new AbortController(); const t = setTimeout(()=>ctl.abort(), 3500);
    const r = await fetch(url, {headers:{Range:'bytes=0-63'}, mode:'cors', signal:ctl.signal});
    clearTimeout(t);
    const b = new Uint8Array(await r.arrayBuffer());
    if(b.length < 8) return null;
    const u8  = (...s)=>s.every((x,i)=>b[i]===x);
    const txt = (s,off=0)=>[...s].every((c,i)=>b[off+i]===c.charCodeAt(0));
    if(u8(0xFF,0xD8,0xFF)) return {type:'image', mime:'image/jpeg'};
    if(u8(0x89,0x50,0x4E,0x47)) return {type:'image', mime:'image/png'};
    if(txt('GIF8')) return {type:'image', mime:'image/gif'};
    if(txt('RIFF') && txt('WEBP',8)) return {type:'image', mime:'image/webp'};
    if(txt('RIFF') && txt('WAVE',8)) return {type:'audio', mime:'audio/wav'};
    if(txt('OggS')) return {type:'audio', mime:'audio/ogg'};
    if(txt('ftyp',4)){
      const brand = String.fromCharCode(b[8],b[9],b[10],b[11]);
      if(/^(avif|avis|heic|mif1|msf1)/.test(brand)) return {type:'image', mime:'image/avif'};
      if(/^M4A/.test(brand)) return {type:'audio', mime:'audio/mp4'};
      return {type:'video', mime:'video/mp4'};
    }
    if(u8(0x1A,0x45,0xDF,0xA3)) return {type:'video', mime:'video/webm'};
    if(txt('ID3') || (b[0]===0xFF && (b[1]&0xE0)===0xE0)) return {type:'audio', mime:'audio/mpeg'};
    return null;
  }catch(e){ return null; }
}

    const t = setTimeout(()=>{ img.src=''; done(null); }, 5000);
    img.onload  = ()=>{ clearTimeout(t); done(img.naturalWidth>0 ? {type:'image', width:img.naturalWidth, height:img.naturalHeight} : null); };
    img.onerror = ()=>{ clearTimeout(t); done(null); };
    img.src = url;
  });
}

/* L4b — giải mã được như video không? videoWidth=0 + có duration ⇒ audio */
function probeVideo(url){
  return new Promise(res=>{
    const v = document.createElement('video');
    v.preload = 'metadata'; v.muted = true;
    const done = r => { v.onloadedmetadata = v.onerror = null; v.removeAttribute('src'); try{v.load();}catch(e){} res(r); };
    const t = setTimeout(()=>done(null), 5000);
    v.onloadedmetadata = ()=>{
      clearTimeout(t);
      if(v.videoWidth > 0) done({type:'video', width:v.videoWidth, height:v.videoHeight, duration:v.duration});
      else if(!isNaN(v.duration) && v.duration > 0) done({type:'audio', duration:v.duration});
      else done(null);
    };
    v.onerror = ()=>{ clearTimeout(t); done(null); };
    v.src = url;
  });
}

/* Orchestrator — trả về { type, mime, method, confidence, width, height, duration, chain } */
async function detectMedia(url, onStep){
  const step = (k,v,c)=> onStep && onStep(k,v,c);
  const ext = extOf(url);
  const extType = ext ? (EXT_TYPES.image.includes(ext)?'image':EXT_TYPES.video.includes(ext)?'video':EXT_TYPES.audio.includes(ext)?'audio':null) : null;
  step('sniff · extension', ext ? `${ext} → ${extType||'?'}` : 'no extension', extType?'ok':'warn');

  const [head, magic, img, vid] = await Promise.all([
    sniffTimeout(probeHead(url), 4200),  sniffTimeout(probeMagic(url), 4200),
    sniffTimeout(probeImage(url), 6000), sniffTimeout(probeVideo(url), 6000),
  ]);

  const chain = [
    {k:'file extension',    v: ext,  ok: !!extType},
    {k:'HTTP content-type', v: head  ? head.mime : null,               ok: !!head},
    {k:'magic bytes',       v: magic ? magic.mime : null,              ok: !!magic},
    {k:'decoder <img>',     v: img   ? `${img.width}×${img.height}` : null, ok: !!img},
    {k:'decoder <video>',   v: vid   ? (vid.width ? `${vid.width}×${vid.height}` : 'audio track') : null, ok: !!vid},
  ];
  chain.forEach(c => step('sniff · '+c.k, c.v || '—', c.ok ? 'ok' : 'warn'));

  const first = head || magic || img || vid;
  const type = first ? first.type : extType;   // header > magic > decoder > extension
  const method = head ? 'HTTP Content-Type' : magic ? 'Magic bytes'
    : img ? 'Decoder <img>' : vid ? 'Decoder <video>' : extType ? 'File extension' : '—';
  const confidence = (head||magic) ? 'high' : (img||vid) ? 'confirmed' : extType ? 'low' : 'none';

  step('sniff · verdict', type ? `${type.toUpperCase()} — ${method}` : 'UNKNOWN', type ? 'ok' : 'err');
  return {
    type, method, confidence,
    mime: (head||magic) ? (head||magic).mime : null,
    width:  img ? img.width  : vid ? vid.width  : undefined,
    height: img ? img.height : vid ? vid.height : undefined,
    duration: vid ? vid.duration : undefined,
    chain,
  };
}

/* Poster frame cho video direct link (thumbnail nhỏ, an toàn localStorage) */
function videoPoster(url, at=1.0){
  return new Promise(res=>{
    const v = document.createElement('video');
    v.crossOrigin = 'anonymous'; v.muted = true; v.preload = 'metadata';
    const done = r => { clearTimeout(t); v.onloadeddata=v.onseeked=v.onerror=null; v.removeAttribute('src'); try{v.load();}catch(e){} res(r); };
    const t = setTimeout(()=>done(null), 7000);
    v.onloadeddata = ()=>{ try{ v.currentTime = Math.min(at, Math.max(0.1,(v.duration||2)/3)); }catch(e){ done(null); } };
    v.onseeked = ()=>{
      try{
        const scale = Math.min(1, 480 / (v.videoWidth || 480));
        const c = document.createElement('canvas');
        c.width = Math.round((v.videoWidth||480)*scale); c.height = Math.round((v.videoHeight||270)*scale);
        c.getContext('2d').drawImage(v, 0, 0, c.width, c.height);
        done(c.toDataURL('image/jpeg', 0.6));
      }catch(e){ done(null); }   // canvas tainted → server không cho CORS
    };
    v.onerror = ()=>done(null);
    v.src = url;
  });
}

let CURRENT = null; // media info đang hiển thị

async function analyze(rawUrl){
  const url = rawUrl.trim();
  const zone = $('#analyzeZone'), lines = $('#consoleLines'), bar = $('#analyzeBar');
  const btn = $('#analyzeBtn');
  $('#errBox').classList.remove('on');
  $('#resultZone').classList.remove('on');
  if(!url){ toast('Paste a link first.', 'info'); return; }

  let info = null, host = '', provider = null;
  try{ host = new URL(url).hostname.replace(/^www\./,''); }catch(e){}
  provider = findProvider(url);

  lines.innerHTML = ''; bar.style.width = '4%';
  zone.classList.add('on'); btn.disabled = true;
  btn.innerHTML = '<span class="spin" style="display:inline-block;width:14px;height:14px;border:2px solid rgba(255,255,255,.35);border-top-color:#fff;border-radius:50%;animation:rot .7s linear infinite"></span>Analyzing';

  const line = (txt, val, cls='') => {
    const d = document.createElement('div');
    d.className = 'cline ' + cls;
    d.innerHTML = `<span class="p">›</span><span>${esc(txt)}</span><span class="v">${esc(val||'')}</span>`;
    lines.appendChild(d); icons();
  };
  const wait = ms => new Promise(r=>setTimeout(r,ms));

  line('parsing url', host || 'invalid', host ? 'ok' : 'err');
  bar.style.width = '18%';
  await wait(320);

  if(!provider && !(provider = tryGeneric(url))){
    line('matching provider', 'no handler', 'err');
    bar.style.width = '100%';
    await wait(500);
    finishFail(); return;
  }
  line('matching provider', provider.name, 'ok');
  bar.style.width = '36%';

  const generic = provider.id === 'web';
  let type, sniff = null;

  if(generic){
    /* Direct link → sniff thật: ảnh, video hay audio? (4 lớp kiểm chứng) */
    line('sniffing content type', 'extension → header → magic bytes → decoder', 'warn');
    sniff = await detectMedia(url, (k,v,c)=>line(k,v,c));
    type = sniff && sniff.type;
    bar.style.width = '58%';
    if(!type){
      line('sniff · verdict', 'no decodable media found', 'err');
      bar.style.width = '100%';
      await wait(500);
      finishFail(); return;   /* URL trỏ tới trang web, không phải media */
    }
  } else {
    type = provider.type ? provider.type(url) : 'video';
  }

  /* Metadata công khai qua oEmbed (bỏ qua với direct link) */
  const metaP = generic ? Promise.resolve(null) : fetchOEmbed(url).catch(()=>null);
  const [meta] = await Promise.all([metaP, wait(650)]);
  bar.style.width = '66%';

  if(!generic){
    line('fetching public metadata (oEmbed)', meta && meta.title ? 'ok' : 'unavailable → backend',
      meta && meta.title ? 'ok' : 'warn');
  }
  await wait(380);
  bar.style.width = '84%';

  const formats = provider.formats ? provider.formats({url, type}) : [];
  line('resolving formats', `${formats.length} available`, 'ok');
  bar.style.width = '100%';
  await wait(480);

  /* Dựng MediaInfo */
  let title = meta && meta.title ? meta.title : null;
  let author = meta && meta.author_name ? meta.author_name : null;
  let thumb = meta && (meta.thumbnail_url || meta.url) ? (meta.thumbnail_url || meta.url) : null;
  let duration = null;

  if(generic && sniff){
    if(sniff.type === 'image'){
      thumb = url;
    } else if(sniff.type === 'video'){
      const p = await videoPoster(url);
      thumb = p || picsum('video-' + url, 800, 640);
      line('capturing poster frame', p ? 'ok' : 'unavailable', p ? 'ok' : 'warn');
    } else {
      thumb = picsum('audio-' + url, 800, 640);          // artwork cho audio
    }
    if(sniff.duration && isFinite(sniff.duration)) duration = sniff.duration;
    try{ const fn = new URL(url).pathname.split('/').filter(Boolean).pop(); if(fn) title = decodeURIComponent(fn); }catch(e){}
  }

  if(provider.id === 'youtube'){
    const id = ytId(url);
    if(id && !thumb) thumb = `https://i.ytimg.com/vi/${id}/hqdefault.jpg`;
  }
  if(!thumb){ thumb = picsum(provider.id + '-' + url, 800, 640); }
  if(!title){ title = `${provider.name} ${type === 'album' ? 'post' : type} · ${host}`; }
  if(meta && meta.duration) duration = meta.duration > 5000 ? meta.duration/1000 : meta.duration;

  const live = !!meta;
  info = {
    id: uid(), url, provider, host, type, title, author, thumb,
    duration: fmtDuration(duration), formats, live, generic,
    embed: provider.id === 'youtube' ? (ytId(url) ? `https://www.youtube.com/embed/${ytId(url)}?autoplay=1` : null)
         : provider.id === 'vimeo'  ? (url.match(/vimeo\.com\/(\d+)/) ? `https://player.vimeo.com/video/${url.match(/vimeo\.com\/(\d+)/)[1]}?autoplay=1` : null)
         : null,
    gallery: type === 'album' ? [1,2,3,4].map(n => ({ n, thumb: picsum(provider.id+'-'+url+'-'+n, 400, 400) })) : null,
    backendNote: !live || provider.mode === 'backend',
    sniff
  };

  await wait(300);
  zone.classList.remove('on');
  btn.disabled = false;
  btn.innerHTML = '<i data-lucide="zap" width="16"></i>Analyze'; icons();
  renderResult(info);
}

/* ---------- Download queue ---------- */
/* Progress mô phỏng pipeline processing; khi xong, mọi file khả dụng
   từ trình duyệt (ảnh công khai, dataURL) được tải thật; direct
   video/audio bị chặn CORS được giao cho downloader của trình duyệt;
   phần còn lại nhận manifest JSON để backend stream. */
const jobs = [];
function enqueueForMedia(m, fmts, singleName, preDeliv){
  const f = fmts[0] || {label:'file', kind:'file'};
  const srcUrl = m.sniff ? m.url : null;
  const d = preDeliv || [{name:singleName, url: srcUrl || (f.kind==='image' ? m.thumb : null), fmt:f}];
  addJob({
    name: singleName,
    deliverables: d,
    meta: { title:m.title, platform:m.provider.name, url:m.url, format:f.label, type:f.kind, thumb:m.thumb }
  });
}
function addJob(job){
  job.id = uid(); job.progress = 0; job.status = 'active';
  jobs.unshift(job);
  $('#qBadge').style.display = 'grid';
  renderQueue(); openQueue();
  runJob(job);
}
function runJob(job){
  const tick = setInterval(()=>{
    job.progress = Math.min(100, job.progress + 4 + Math.random()*9);
    renderQueue();
    if(job.progress >= 100){
      clearInterval(tick);
      deliver(job);
    }
  }, 130);
}
async function tryFetch(url){
  if(!url) return null;
  try{
    const ctl = new AbortController(); const t = setTimeout(()=>ctl.abort(), 9000);
    const r = await fetch(url, {mode:'cors', signal:ctl.signal});
    clearTimeout(t);
    if(!r.ok) return null;
    return await r.blob();
  }catch(e){ return null; }
}
function saveBlob(blob, name){
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = name;
  document.body.appendChild(a); a.click(); a.remove();
  setTimeout(()=>URL.revokeObjectURL(a.href), 4000);
}
function anchorDownload(url, name){
  const a = document.createElement('a');
  a.href = url; a.download = name; a.target = '_blank'; a.rel = 'noopener';
  document.body.appendChild(a); a.click(); a.remove();
}
function manifestText(job, d){
  return JSON.stringify({
    mediahub_manifest: '1.0',
    notice: 'Demo build: this manifest describes the requested media. Wire the matching provider to your backend to stream the actual file.',
    requested_file: d.name,
    source_url: job.meta.url,
    platform: job.meta.platform,
    format: d.fmt ? d.fmt.label : job.meta.format,
    content_type: d.fmt ? d.fmt.kind : job.meta.type,
    title: job.meta.title,
    generated_at: new Date().toISOString()
  }, null, 2);
}
async function deliver(job){
  const files = [];
  let i = 0;
  for(const d of job.deliverables){
    const blob = await tryFetch(d.url);
    if(blob){ files.push({name:d.name, blob}); continue; }
    /* Server chặn CORS → file direct đơn lẻ: giao cho trình duyệt tải native */
    if(d.url && /^https?:/i.test(d.url) && job.deliverables.length === 1 && !job.zipName){
      anchorDownload(d.url, d.name);
      job.status = 'done';
      addHistory(job.meta, 1);
      toast('Server blocked direct fetch — handed off to your browser downloader.', 'info');
      renderQueue();
      return;
    }
    files.push({name:d.name.replace(/\.[a-z0-9]+$/i,'') + `.manifest-${++i}.json`, blob:new Blob([manifestText(job,d)],{type:'application/json'})});
  }
  try{
    if(files.length === 1){
      saveBlob(files[0].blob, files[0].name);
    } else {
      const zip = new JSZip();
      for(const f of files) zip.file(f.name, f.blob);
      zip.file('_NOTICE.txt',
`MediaHub Downloader — bundle notice
Files ending in .manifest.json describe media that requires the
backend provider to stream (video/audio streams are not fetchable
from the browser). Public images were embedded directly.
Source: ${job.meta.url}`);
      const blob = await zip.generateAsync({type:'blob'});
      saveBlob(blob, job.zipName || 'mediahub-download.zip');
    }
    job.status = 'done';
    addHistory(job.meta, files.length);
    toast(`Saved: ${job.name}`, 'ok');
  }catch(e){
    job.status = 'fail';
    toast('Download failed — please try again.', 'err');
  }
  renderQueue();
}

      <a href="#/" class="btn-accent" style="text-decoration:none"><i data-lucide="arrow-right" width="15"></i>Download something</a>
    </div>`; icons(); return;
  }
  body.innerHTML = `
    <div class="hist-actions">
      <h2>${h.length} item${h.length>1?'s':''}</h2>
      <button class="btn-danger" id="clearAll"><i data-lucide="trash-2"></i>Clear All</button>
    </div>
    <div class="glass" style="padding:8px 14px; margin-bottom:60px">
      ${h.map(e => `
      <div class="hrow" data-id="${e.id}">
        ${e.thumb ? `<img class="hthumb" src="${esc(e.thumb)}" loading="lazy" onerror="this.outerHTML='<div class=\\'hthumb ph\\'><i data-lucide=\\'image\\'></i></div>'"/>`
                  : `<div class="hthumb ph"><i data-lucide="file-video"></i></div>`}
        <div><div class="ht">${esc(e.title)}</div><div class="hm">${esc(e.platform)}${e.files>1?` · ${e.files} files`:''}</div></div>
        <span class="hbadge">${esc(e.format)}</span>
        <span class="hdate">${new Date(e.date).toLocaleDateString(undefined,{month:'short',day:'numeric'})} ${new Date(e.date).toLocaleTimeString(undefined,{hour:'2-digit',minute:'2-digit'})}</span>
        <div class="hacts">
          <button class="icon-btn" data-again title="Download again"><i data-lucide="rotate-ccw"></i></button>
          <button class="icon-btn red" data-del title="Remove"><i data-lucide="trash-2"></i></button>
        </div>
      </div>`).join('')}
    </div>`;
  icons();
  $('#clearAll').addEventListener('click', ()=>{
    setHistory([]); renderHistory(); toast('History cleared.', 'ok');
  });
  body.querySelectorAll('[data-del]').forEach(b => b.addEventListener('click', ()=>{
    const row = b.closest('.hrow');
    setHistory(getHistory().filter(x => x.id !== row.dataset.id));
    renderHistory(); toast('Entry removed.', 'info');
  }));
  body.querySelectorAll('[data-again]').forEach(b => b.addEventListener('click', ()=>{
    const e = getHistory().find(x => x.id === b.closest('.hrow').dataset.id);
    if(!e) return;
    const ext = e.type==='audio'?'mp3':e.type==='image'?'jpg':'mp4';
    const direct = e.platform === GENERIC.name;   /* direct link → tải lại đúng URL gốc */
    addJob({
      name: `${slug(e.title)}.${ext}`,
      deliverables:[{name:`${slug(e.title)}.${ext}`, url: direct ? e.url : e.thumb, fmt:{label:e.format, kind:e.type}}],
      meta:{...e, files:1}
    });
    toast('Re-queued for download.', 'ok');
  }));
}

/* ---------- Router ---------- */
const ROUTES = {'':'home','/':'home','/history':'history','/platforms':'platforms','/about':'about','/api-docs':'api-docs'};
function route(){
  let h = location.hash.replace(/^#/,'') || '/';
  const page = ROUTES[h] || ROUTES[h.replace(/\/$/,'')] || null;
  $$('.page').forEach(p => p.classList.remove('on'));
  if(page){
    $('#page-' + page).classList.add('on');
    $$('#navLinks a').forEach(a => a.classList.toggle('on', a.dataset.r === (h==='/'?'/':h.replace(/\/$/,''))));
    if(page === 'history') renderHistory();
    if(page === 'platforms') renderPlatforms();
    window.scrollTo({top:0});
  } else {
    $('#page-home').classList.add('on');
  }
  $('#mNav').classList.remove('open');
  requestAnimationFrame(scanReveal);
}
window.addEventListener('hashchange', route);
 $('#menuBtn').addEventListener('click', ()=> $('#mNav').classList.toggle('open'));

/* ---------- Quick type detector ---------- */
 $('#detectForm').addEventListener('submit', async e=>{
  e.preventDefault();
  const url = $('#detectInput').value.trim();
  const out = $('#detectOut');
  if(!url){ toast('Paste a URL to detect.', 'info'); return; }
  out.classList.add('on');
  out.innerHTML = `<div class="pbar"><i style="width:35%"></i></div>
    <div class="mono" style="color:var(--dim);font-size:11px;margin-top:10px">probing: extension → header → magic bytes → decoder…</div>`;
  const r = await detectMedia(url);
  const t = r.type || 'unknown';
  const dims = r.width ? ` ${r.width}×${r.height}` : '';
  const dur  = (r.duration && isFinite(r.duration)) ? ` · ${fmtDuration(r.duration)}` : '';
  const label = t==='image' ? 'IMG' : t==='video' ? 'VID' : t==='audio' ? 'AUD' : '?';
  out.innerHTML = `
    <div class="dt-verdict ${t}">
      <span class="dv">${label}</span>
      <div style="min-width:0">
        <h4>${t==='unknown' ? 'Unknown media type' : esc(t.toUpperCase()) + esc(dims) + esc(dur)}</h4>
        <p>${esc(r.method)} · confidence: ${esc(r.confidence)}${r.mime ? ' · ' + esc(r.mime) : ''}</p>
      </div>
    </div>
    <div class="dt-chain">
      ${r.chain.map(c=>`<div class="dc ${c.ok?'y':'n'}"><b>${esc(c.k)}</b><b>${esc(c.v || (c.ok ? '✓' : '✗ / blocked'))}</b></div>`).join('')}
    </div>`;
});
